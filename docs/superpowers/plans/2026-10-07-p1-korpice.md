# P1 — Korpice i sedmični ciklus: plan implementacije

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** E piljara dobija korpicu — pakovanje domaćih proizvoda sa fiksnom cijenom čiji se sadržaj generiše iznova svake sedmice, prodaje se kroz postojeći checkout i pakuje po listi iz admin panela.

**Architecture:** Korpica je `Product` sa `Vrsta = Korpica`, pa se u korpi, narudžbi i na checkoutu ponaša kao svaki drugi artikal — `Order`, `OrderItem` i tok naručivanja se ne diraju. Sadržaj živi u zasebnim tabelama (`SedmicniSadrzaj` + `StavkaSadrzaja`) sa statusom kao state machine (`Nacrt → Objavljen → Zakljucan`). Generator sastavlja prijedlog iz sezone, zaliha i prioriteta; admin ga doradi i objavi.

**Tech Stack:** ASP.NET Core 9, EF Core 9 + PostgreSQL, MediatR (CQRS), FluentValidation, xUnit + SQLite in-memory; Next.js 15 (App Router), TypeScript, Tailwind v4, TanStack Query.

**Spec:** `docs/superpowers/specs/2026-10-07-p1-korpice-design.md`

## Global Constraints

- **Grana:** `korpice-p1`, izvedena iz `redizajn-opcija3` (ne iz `main`-a).
- **Jezik:** svi korisnički tekstovi i poruke grešaka na srpskom, ijekavica. Imena entiteta, polja i komandi takođe na srpskom — prati postojeće (`Magacin`, `ZatvorenaU`, `GuestOrderRequest.TrajanjeMinuta`).
- **Backend komande iz `backend/`, frontend iz `frontend/`.**
- **Svaki endpoint vraća kovertu `{ data, error, meta }`** — koristi `Envelope`/`EnvelopeList` iz `ApiControllerBase`, nikad ručno sastavljen odgovor.
- **Greške se bacaju kao domenski izuzeci** (`NotFoundException` → 404, `ConflictException` → 409, `DomainException` → 400). Middleware ih mapira; ne praviti odgovor u kontroleru.
- **Enumi se u bazu pišu kao string:** `e.Property(x => x.Status).HasConversion<string>().HasMaxLength(16);`
- **MediatR i FluentValidation se otkrivaju skeniranjem assemblyja** — nema registracije po handleru.
- **Frontend nikad ne zove `fetch` direktno** — samo kroz `frontend/src/lib/api/endpoints.ts`. Tipovi domena u `types/domain.ts`, oblici zahtjeva u `lib/api/contracts.ts`.
- **Mock (`frontend/src/server/mock/`) se NE dopunjava.** Korpice se razvijaju protiv pravog backenda; `frontend/.env.local` mora imati `NEXT_PUBLIC_API_BASE_URL=http://localhost:5080`.
- **Frontend provjera je `npx tsc --noEmit` + `npm run lint` + obavezno `npm run build`.** Zeleni `tsc` i `lint` nisu dokaz da se build ne ruši — `useSearchParams()` bez `<Suspense>` omotača obara prerender.
- **`KoeficijentVrijednosti` = 1.15**, čita se iz konfiguracije (`Korpice:KoeficijentVrijednosti`), ne iz konstante.
- **Minimalan broj artikala:** Mala 5, Srednja 7, Velika 9.
- **Rok naručivanja:** `datum isporuke − Korpice:RokSati` (default 48).

```csharp
            e.Property(x => x.TipKorpice).HasConversion<string>().HasMaxLength(16);
            e.Property(x => x.Velicina).HasConversion<string>().HasMaxLength(16);

            // SQLite koji nema tip niza. Jedan oblik koji radi na oba providera.
            e.Property(x => x.Mjeseci)
                .HasConversion(
                    v => string.Join(',', v),
                    v => string.IsNullOrWhiteSpace(v)
                        ? Array.Empty<int>()
                        : v.Split(',', StringSplitOptions.RemoveEmptyEntries).Select(int.Parse).ToArray(),
                    new ValueComparer<int[]>(
                        (a, c) => a!.SequenceEqual(c!),
                        v => v.Aggregate(0, (h, x) => HashCode.Combine(h, x)),
                        v => v.ToArray()))
                .HasMaxLength(60)
                .HasDefaultValue(Array.Empty<int>());

            e.HasIndex(x => x.Vrsta);
```

Dodaj na vrh fajla: `using Microsoft.EntityFrameworkCore.ChangeTracking;`

`ValueComparer` nije opcion — bez njega EF ne primjećuje izmjenu unutar niza i tiho ne upisuje promjenu.

```csharp
    public static readonly IReadOnlyDictionary<StatusSadrzaja, StatusSadrzaja[]> Prelazi =
        new Dictionary<StatusSadrzaja, StatusSadrzaja[]>
        {
            [StatusSadrzaja.Nacrt] = [StatusSadrzaja.Objavljen],
            [StatusSadrzaja.Objavljen] = [StatusSadrzaja.Zakljucan],
            [StatusSadrzaja.Zakljucan] = [],
        };

    public bool MozePreci(StatusSadrzaja u) =>
        Prelazi.TryGetValue(Status, out var dozvoljeni) && dozvoljeni.Contains(u);

    /// <summary>Mijenja status uz provjeru. Baca ConflictException na nedozvoljen prelaz.</summary>
    public void Predji(StatusSadrzaja u)
    {
        if (!MozePreci(u))
            throw new ConflictException($"nedozvoljen_prelaz_sadrzaja: {Status} → {u}");

        Status = u;
        if (u == StatusSadrzaja.Objavljen) ObjavljenU = DateTime.UtcNow;
        if (u == StatusSadrzaja.Zakljucan) ZakljucanU = DateTime.UtcNow;
    }
```

```csharp
using System.Globalization;
using EPiljara.Application.Common.Interfaces;
using EPiljara.Domain.Entities;
using EPiljara.Domain.Enums;
using MediatR;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Options;

namespace EPiljara.Application.Features.Korpice;

public record RezultatGenerisanjaDto
{
    public Guid SadrzajId { get; init; }
    public Guid KorpicaId { get; init; }
    public string KorpicaNaziv { get; init; } = string.Empty;
    public int BrojArtikala { get; init; }
    public decimal Vrijednost { get; init; }
    public decimal CiljGenerisanja { get; init; }

    /// <summary>Popunjeno kad bazen nije mogao ispuniti minimum. Nije greška — admin dorađuje.</summary>
    public string? Upozorenje { get; init; }
}

/// <summary>
/// Sastavlja PRIJEDLOG sadržaja za zadatu sedmicu. Rezultat je uvijek Nacrt —
/// admin je posljednja instanca. Korpice koje za tu sedmicu već imaju sadržaj se preskaču.
/// </summary>
public record GenerisiSedmicniSadrzajCommand(int Godina, int Sedmica, Guid? KorpicaId)
    : IRequest<IReadOnlyList<RezultatGenerisanjaDto>>;

public class GenerisiSedmicniSadrzajHandler(IAppDbContext db, IOptions<KorpiceOpcije> opcije)
    : IRequestHandler<GenerisiSedmicniSadrzajCommand, IReadOnlyList<RezultatGenerisanjaDto>>
{
    private readonly KorpiceOpcije _o = opcije.Value;

    public async Task<IReadOnlyList<RezultatGenerisanjaDto>> Handle(
        GenerisiSedmicniSadrzajCommand c, CancellationToken ct)
    {
        var korpice = await db.Products
            .Where(p => p.Vrsta == VrstaProizvoda.Korpica && p.IsAvailable)
            .Where(p => c.KorpicaId == null || p.Id == c.KorpicaId)
            .ToListAsync(ct);

        var vecPostoje = await db.SedmicniSadrzaji
            .Where(s => s.Godina == c.Godina && s.Sedmica == c.Sedmica)
            .Select(s => s.KorpicaId)
            .ToListAsync(ct);

        var veze = await db.KorpicaKategorije.ToListAsync(ct);
        var mjesec = MjesecSedmice(c.Godina, c.Sedmica);
        var rezultati = new List<RezultatGenerisanjaDto>();

        foreach (var korpica in korpice.Where(k => !vecPostoje.Contains(k.Id)))
        {
            var kategorije = veze
                .Where(v => v.TipKorpice == korpica.TipKorpice)
                .Select(v => v.CategoryId)
                .ToHashSet();

            // Mjeseci su CSV u bazi, pa se sezona filtrira u memoriji nakon učitavanja.
            var bazen = (await db.Products
                    .Where(p => p.Vrsta == VrstaProizvoda.Obican
                                && p.MozeUKorpicu
                                && p.IsAvailable
                                && p.Stock > 0
                                && kategorije.Contains(p.CategoryId))
                    .ToListAsync(ct))
                .Where(p => p.Mjeseci.Length == 0 || p.Mjeseci.Contains(mjesec))
                .OrderBy(p => p.PrioritetUKorpici)
                .ThenBy(_ => Guid.NewGuid())      // nasumično unutar istog prioriteta
                .ToList();

            var minimum = _o.MinimalanBrojArtikala.GetValueOrDefault(
                korpica.Velicina ?? VelicinaKorpice.Mala, 5);
            var cilj = korpica.Price * _o.KoeficijentVrijednosti;

            // Budžet po stavci dok se ne ispuni minimum. Bez ovoga prvi artikal
            // pojede cio cilj i korpica ispadne sa jednom stavkom.
            var poStavci = cilj / Math.Max(minimum, 1);

            var sadrzaj = new SedmicniSadrzaj
            {
                KorpicaId = korpica.Id,
                Godina = c.Godina,
                Sedmica = c.Sedmica,
                Status = StatusSadrzaja.Nacrt,
            };

            var vrijednost = 0m;
            foreach (var artikal in bazen)
            {
                if (vrijednost >= cilj * 0.95m && sadrzaj.Stavke.Count >= minimum) break;
                if (artikal.Price <= 0) continue;

                var preostalo = cilj - vrijednost;
                if (preostalo <= 0) break;

                var budzet = sadrzaj.Stavke.Count < minimum
                    ? Math.Min(poStavci, preostalo)
                    : preostalo;

                var kolicina = Zaokruzi(budzet / artikal.Price, artikal.Unit);
                if (kolicina > artikal.Stock) kolicina = Zaokruzi(artikal.Stock, artikal.Unit);
                if (kolicina <= 0) continue;

                sadrzaj.Stavke.Add(new StavkaSadrzaja
                {
                    ProductId = artikal.Id,
                    Kolicina = kolicina,
                    JedinicnaCijena = artikal.Price,
                });
                vrijednost += kolicina * artikal.Price;
            }

            db.SedmicniSadrzaji.Add(sadrzaj);

            rezultati.Add(new RezultatGenerisanjaDto
            {
                SadrzajId = sadrzaj.Id,
                KorpicaId = korpica.Id,
                KorpicaNaziv = korpica.Name,
                BrojArtikala = sadrzaj.Stavke.Count,
                Vrijednost = vrijednost,
                CiljGenerisanja = cilj,
                Upozorenje = sadrzaj.Stavke.Count < minimum
                    ? $"U bazenu je premalo artikala: {sadrzaj.Stavke.Count} od {minimum}. Dopuni sadržaj ručno."
                    : null,
            });
        }

        await db.SaveChangesAsync(ct);
        return rezultati;
    }

    /// <summary>Korak je 0.1 za težinu i zapreminu, 1 za komade.</summary>
    private static decimal Zaokruzi(decimal kolicina, Unit jedinica)
    {
        if (jedinica == Unit.Kom) return Math.Floor(kolicina);
        return Math.Floor(kolicina * 10m) / 10m;
    }

    /// <summary>Mjesec u koji pada četvrtak zadate ISO sedmice — četvrtak je uvijek u „pravom" mjesecu.</summary>
    private static int MjesecSedmice(int godina, int sedmica) =>
        ISOWeek.ToDateTime(godina, sedmica, DayOfWeek.Thursday).Month;
}
```

```csharp
namespace EPiljara.Application.Features.Korpice;

/// <summary>
/// Nacrt → Objavljen. Ponovo snima jedinične cijene i zamrzava CiljnuVrijednost,
/// koja je od tog trenutka budžet kupčevog editora (P2).
/// </summary>

public class ObjaviSadrzajHandler(IAppDbContext db) : IRequestHandler<ObjaviSadrzajCommand, Guid>

    {
        var sadrzaj = await db.SedmicniSadrzaji
            .Include(s => s.Stavke).ThenInclude(st => st.Product)
            .FirstOrDefaultAsync(s => s.Id == c.SadrzajId, ct)
            ?? throw new NotFoundException("Sadržaj korpice nije pronađen.");

        if (sadrzaj.Stavke.Count == 0)
            throw new DomainException("Prazan sadržaj se ne može objaviti.");

        foreach (var stavka in sadrzaj.Stavke)
            stavka.JedinicnaCijena = stavka.Product?.Price ?? stavka.JedinicnaCijena;

        sadrzaj.CiljnaVrijednost = sadrzaj.Stavke.Sum(s => s.Kolicina * s.JedinicnaCijena);
        sadrzaj.Predji(StatusSadrzaja.Objavljen);

        await db.SaveChangesAsync(ct);
        return sadrzaj.Id;
    }
}
```

```csharp
public static class SedmicaHelper
{
    public static (int Godina, int Sedmica) Tekuca(DateTime? sada = null)
    {
        var d = sada ?? DateTime.UtcNow;
        return (ISOWeek.GetYear(d), ISOWeek.GetWeekOfYear(d));
    }

    /// <summary>Petak zadate ISO sedmice — dogovoreni dan isporuke korpica.</summary>
    public static DateTime DatumIsporuke(int godina, int sedmica) =>
        ISOWeek.ToDateTime(godina, sedmica, DayOfWeek.Friday);

    public static string Slug(TipKorpice t) => t switch
    {
        TipKorpice.Povrce => "povrce",
        TipKorpice.Voce => "voce",
        TipKorpice.Mlijecna => "mlijecna",
        TipKorpice.Suhomesnata => "suhomesnata",
        TipKorpice.Domacinska => "domacinska",
        _ => "povrce",
    };

    public static TipKorpice? IzSluga(string? s) => s?.Trim().ToLowerInvariant() switch
    {
        "povrce" => TipKorpice.Povrce,
        "voce" => TipKorpice.Voce,
        "mlijecna" => TipKorpice.Mlijecna,
        "suhomesnata" => TipKorpice.Suhomesnata,
        "domacinska" => TipKorpice.Domacinska,
        _ => null,

    /// NE zove se ToDto: <c>MappingExtensions.ToDto(this Product)</c> već postoji i vraća
    /// ProductDto. Dvije ekstenzije istog imena nad istim tipom čine svaki poziv dvosmislenim.
    /// </summary>
    public static KorpicaDto ToKorpicaDto(this Product p) => new()
    {
        Id = p.Id,
        Naziv = p.Name,
        Slug = p.Slug,
        Tip = Slug(p.TipKorpice ?? TipKorpice.Povrce),
        Velicina = (p.Velicina ?? VelicinaKorpice.Mala).ToString().ToLowerInvariant(),
        Cijena = p.Price,
        ImageUrl = p.ImageUrl,
        Opis = p.Description,
        Stock = p.Stock,
        Dostupna = p.IsAvailable && p.Stock > 0,
    };

    public static SadrzajDto ToDto(this SedmicniSadrzaj s) => new()

public record GetKorpiceQuery(string? Tip) : IRequest<IReadOnlyList<KorpicaDto>>;

public class GetKorpiceHandler(IAppDbContext db)
    : IRequestHandler<GetKorpiceQuery, IReadOnlyList<KorpicaDto>>
{
    public async Task<IReadOnlyList<KorpicaDto>> Handle(GetKorpiceQuery q, CancellationToken ct)
    {
        var tip = SedmicaHelper.IzSluga(q.Tip);

        var lista = await db.Products
            .Where(p => p.Vrsta == VrstaProizvoda.Korpica)
            .Where(p => tip == null || p.TipKorpice == tip)
            .OrderBy(p => p.TipKorpice).ThenBy(p => p.Velicina)
            .ToListAsync(ct);

        return lista.Select(p => p.ToKorpicaDto()).ToList();
    }
}

            ?? throw new NotFoundException("Korpica nije pronađena.");

                s.KorpicaId == korpica.Id && s.Godina == godina && s.Sedmica == sedmica
                && s.Status != StatusSadrzaja.Nacrt, ct)

public class GetTekucaSedmicaHandler(IOptions<KorpiceOpcije> opcije)

            RokNarucivanja = rok,

    }
```

```csharp
public record StavkaUnos(Guid ProductId, decimal Kolicina);

/// <summary>Zamjenjuje stavke nacrta u cjelosti. Objavljen sadržaj se ne dira.</summary>
public record IzmijeniNacrtCommand(Guid SadrzajId, IReadOnlyList<StavkaUnos> Stavke)
    : IRequest<SadrzajDto>;

public class IzmijeniNacrtHandler(IAppDbContext db)
    : IRequestHandler<IzmijeniNacrtCommand, SadrzajDto>
{
    public async Task<SadrzajDto> Handle(IzmijeniNacrtCommand c, CancellationToken ct)
    {
        var sadrzaj = await db.SedmicniSadrzaji
            .Include(s => s.Stavke)
            .FirstOrDefaultAsync(s => s.Id == c.SadrzajId, ct)
            ?? throw new NotFoundException("Sadržaj korpice nije pronađen.");

        if (sadrzaj.Status != StatusSadrzaja.Nacrt)
            throw new ConflictException("sadrzaj_nije_nacrt: objavljen sadržaj se ne može mijenjati.");

        var idjevi = c.Stavke.Select(s => s.ProductId).Distinct().ToList();
        var artikli = await db.Products.Where(p => idjevi.Contains(p.Id)).ToListAsync(ct);

        if (artikli.Count != idjevi.Count)
            throw new NotFoundException("Jedan ili više artikala nije pronađen.");

        if (artikli.Any(a => a.Vrsta == VrstaProizvoda.Korpica))
            throw new DomainException("Korpica ne može biti stavka druge korpice.");

        if (c.Stavke.Any(s => s.Kolicina <= 0))
            throw new DomainException("Količina mora biti veća od nule.");

        db.StavkeSadrzaja.RemoveRange(sadrzaj.Stavke);
        sadrzaj.Stavke.Clear();

        foreach (var unos in c.Stavke)
        {
            var artikal = artikli.First(a => a.Id == unos.ProductId);
            sadrzaj.Stavke.Add(new StavkaSadrzaja
            {
                ProductId = artikal.Id,
                Kolicina = unos.Kolicina,
                JedinicnaCijena = artikal.Price,
            });
        }

        await db.SaveChangesAsync(ct);
        return sadrzaj.ToDto();
    }
}

    {

            .Include(s => s.Product).ThenInclude(p => p!.Producer)
            .Include(s => s.SedmicniSadrzaj)

                        && s.SedmicniSadrzaj.Sedmica == q.Sedmica
                        && s.SedmicniSadrzaj.Status != StatusSadrzaja.Nacrt)
            .ToListAsync(ct);

        return stavke
            .GroupBy(s => s.ProductId)
            .Select(g =>
            {
                var p = g.First().Product;
                return new StavkaPakerskeDto
                {
                    Artikal = p?.Name ?? string.Empty,
                    Kolicina = g.Sum(x => x.Kolicina),
                    Jedinica = (p?.Unit ?? Unit.Kom).ToApi(),
                    Proizvodjac = p?.Producer?.Name ?? string.Empty,
                    Selo = p?.Producer?.Village ?? string.Empty,
                };
            })
            .OrderBy(r => r.Proizvodjac).ThenBy(r => r.Artikal)
            .ToList();
    }
```

```csharp
using EPiljara.Application.Common.Interfaces;
using EPiljara.Domain.Enums;
using MediatR;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Options;

namespace EPiljara.Application.Features.Korpice;

/// <summary>
/// Prebacuje objavljene sadržaje kojima je rok naručivanja prošao u Zakljucan.
/// Od tog trenutka je pakerska lista zamrznuta. <paramref name="Sada"/> postoji
/// radi testova; u radu ostaje null.
/// </summary>
public record ZakljucajSadrzajeCommand(DateTime? Sada = null) : IRequest<int>;

public class ZakljucajSadrzajeHandler(IAppDbContext db, IOptions<KorpiceOpcije> opcije)
    : IRequestHandler<ZakljucajSadrzajeCommand, int>
{
    public async Task<int> Handle(ZakljucajSadrzajeCommand c, CancellationToken ct)
    {
        var sada = c.Sada ?? DateTime.UtcNow;

        var kandidati = await db.SedmicniSadrzaji
            .Where(s => s.Status == StatusSadrzaja.Objavljen)
            .ToListAsync(ct);

        var zakljucani = 0;
        foreach (var s in kandidati)
        {
            var rok = SedmicaHelper.DatumIsporuke(s.Godina, s.Sedmica)
                .AddHours(-opcije.Value.RokSati);
            if (sada <= rok) continue;

            s.Predji(StatusSadrzaja.Zakljucan);
            zakljucani++;
        }

        if (zakljucani > 0) await db.SaveChangesAsync(ct);
        return zakljucani;
    }
}
```

```csharp
using EPiljara.Application.Features.Korpice;
using MediatR;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

namespace EPiljara.Infrastructure.Services;

/// <summary>
/// Svakih 15 minuta zatvara sadržaje kojima je rok naručivanja prošao.
/// Zaključavanje mora biti automatsko — ako zavisi od toga da se admin sjeti,
/// prije ili kasnije se neće sjetiti.
/// </summary>
public class ZakljucavanjeSadrzajaService(
    IServiceScopeFactory scopeFactory,
    ILogger<ZakljucavanjeSadrzajaService> log) : BackgroundService
{
    private static readonly TimeSpan Interval = TimeSpan.FromMinutes(15);

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                using var scope = scopeFactory.CreateScope();
                var mediator = scope.ServiceProvider.GetRequiredService<ISender>();
                var broj = await mediator.Send(new ZakljucajSadrzajeCommand(), stoppingToken);
                if (broj > 0) log.LogInformation("Zaključano sadržaja korpica: {Broj}", broj);
            }
            catch (Exception ex)
            {
                log.LogError(ex, "Zaključavanje sadržaja korpica nije uspjelo.");
            }

            await Task.Delay(Interval, stoppingToken);
        }
    }
}
```

```csharp
            (TipKorpice.Suhomesnata, "suhomesnati-proizvodi"),
            (TipKorpice.Domacinska, "mlijecni-proizvodi"),
            (TipKorpice.Domacinska, "suhomesnati-proizvodi"),
        };

        foreach (var (tip, slug) in vezeKorpica)
        {
            var kat = kategorije.FirstOrDefault(k => k.Slug == slug);
            if (kat is null) continue;
            db.KorpicaKategorije.Add(new KorpicaKategorija { TipKorpice = tip, CategoryId = kat.Id });
        }

        var definicije = new (TipKorpice Tip, string Naziv, string Opis)[]
        {

            (TipKorpice.Suhomesnata, "suhomesnata korpica", "Pečenica, slanina, kobasica i suvo meso."),
            (TipKorpice.Domacinska, "domaćinska korpica", "Miješano: komad pečenice, slanina, mlijeko i sir."),
        };

        var velicine = new (VelicinaKorpice V, string Prefiks, decimal Faktor, string Clanovi)[]
        {
            (VelicinaKorpice.Mala, "Mala", 1.0m, "Za 1 do 2 člana"),
            (VelicinaKorpice.Srednja, "Srednja", 1.4m, "Za 2 do 4 člana"),
            (VelicinaKorpice.Velika, "Velika", 1.8m, "Za 5 i više članova"),
        };

        var osnovneCijene = new Dictionary<TipKorpice, decimal>
        {
            [TipKorpice.Povrce] = 25m,
            [TipKorpice.Voce] = 28m,
            [TipKorpice.Mlijecna] = 32m,
            [TipKorpice.Suhomesnata] = 45m,
            [TipKorpice.Domacinska] = 48m,
        };

        var prvaKategorija = kategorije.First();
        var prviProizvodjac = proizvodjaci.First();

        foreach (var (tip, naziv, opis) in definicije)
        foreach (var (v, prefiks, faktor, clanovi) in velicine)
        {
            var punNaziv = $"{prefiks} {naziv}";
            db.Products.Add(new Product
            {
                Name = punNaziv,
                Slug = Slugify(punNaziv),
                Price = Math.Round(osnovneCijene[tip] * faktor, 0),
                Unit = Unit.Kom,
                Stock = 40,
                IsAvailable = true,
                Description = $"{opis} {clanovi}.",
                CategoryId = prvaKategorija.Id,
                ProducerId = prviProizvodjac.Id,
                Vrsta = VrstaProizvoda.Korpica,
                TipKorpice = tip,
                Velicina = v,
            });
        }
```

```ts
      query: tip ? { tip } : {},
    }).then((r) => r.data),

  sadrzaj: (slug: string) =>

    apiFetch<Sedmica>("/korpice/sedmica", { auth: false }).then((r) => r.data),

/* ---------- Admin: korpice ---------- */

export const adminKorpiceApi = {
  pregled: (godina: number, sedmica: number) =>
    apiFetch<SadrzajPregled[]>("/admin/korpice/sadrzaj", {
      query: { godina, sedmica },
    }).then((r) => r.data),

  sadrzaj: (id: string) =>
```

```tsx
/** ISO broj sedmice za dati datum — isti račun kao na backendu. */
function isoSedmica(d: Date): { godina: number; sedmica: number } {
  const t = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()));
  t.setUTCDate(t.getUTCDate() + 4 - (t.getUTCDay() || 7));
  const prvi = new Date(Date.UTC(t.getUTCFullYear(), 0, 1));
  const sedmica = Math.ceil(((t.getTime() - prvi.getTime()) / 86400000 + 1) / 7);

}
```

`frontend/src/components/admin/sadrzaj-nacrt-editor.tsx`:

```tsx
"use client";

import { useEffect, useState } from "react";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { adminKorpiceApi, catalogApi } from "@/lib/api/endpoints";

type Red = { productId: string; naziv: string; kolicina: number; cijena: number };

export function SadrzajNacrtEditor({
  sadrzajId,
  godina,
  sedmica,
  onClose,
}: {
  sadrzajId: string;
  godina: number;
  sedmica: number;
  onClose: () => void;
}) {
  const qc = useQueryClient();
  const [redovi, setRedovi] = useState<Red[] | null>(null);

  const { data: pregled } = useQuery({
    queryKey: ["admin", "korpice", godina, sedmica],
    queryFn: () => adminKorpiceApi.pregled(godina, sedmica),
  });

  const { data: artikli } = useQuery({
    queryKey: ["products", "svi"],
    queryFn: () => catalogApi.products({}),
  });

  const { data: sadrzaj } = useQuery({
    queryKey: ["admin", "korpice", "sadrzaj", sadrzajId],
    queryFn: () => adminKorpiceApi.sadrzaj(sadrzajId),
  });

  // Postojeći nacrt puni lokalno stanje jednom; dalje se uređuje lokalno.
  // `redovi === null` znači „još nije učitano" i razlikuje se od praznog nacrta.
  useEffect(() => {
    if (!sadrzaj || redovi !== null) return;
    setRedovi(
      sadrzaj.stavke.map((s) => ({
        productId: s.productId,
        naziv: s.naziv,
        kolicina: s.kolicina,
        cijena: 0, // stvarna cijena dolazi iz liste artikala ispod
      })),
    );
  }, [sadrzaj, redovi]);

  // Cijene se vežu iz kataloga, jer SadrzajDto nosi količinu a ne jediničnu cijenu.
  useEffect(() => {
    if (!artikli || !redovi || redovi.every((r) => r.cijena > 0)) return;
    setRedovi(
      redovi.map((r) => ({
        ...r,
        cijena: artikli.find((a) => a.id === r.productId)?.price ?? 0,
      })),
    );
  }, [artikli, redovi]);

  if (redovi === null) {
    return (
      <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40">
        <p className="rounded-2xl bg-white p-6 text-neutral-500">Učitavanje nacrta…</p>
      </div>
    );
  }

  const ukupno = redovi.reduce((z, r) => z + r.kolicina * r.cijena, 0);
  const cilj = pregled?.find((p) => p.sadrzajId === sadrzajId)?.ciljGenerisanja ?? 0;

  const sacuvaj = useMutation({
    mutationFn: () =>
      adminKorpiceApi.izmijeni(sadrzajId, {
        stavke: redovi.map((r) => ({ productId: r.productId, kolicina: r.kolicina })),
      }),
    onSuccess: () => {
      qc.invalidateQueries({ queryKey: ["admin", "korpice", godina, sedmica] });
      qc.invalidateQueries({ queryKey: ["admin", "korpice", "sadrzaj", sadrzajId] });
      onClose();
    },
  });

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 p-4" role="dialog" aria-modal="true">
      <div className="max-h-[85vh] w-full max-w-2xl overflow-y-auto rounded-2xl bg-white p-6">
        <div className="flex items-center justify-between">
          <h2 className="text-xl font-bold">Izmjena nacrta</h2>
          <button type="button" onClick={onClose} aria-label="Zatvori" className="text-2xl text-neutral-400">
            ×
          </button>
        </div>

        <p className="mt-2 text-sm">
          Vrijednost:{" "}
          <strong className={ukupno > cilj * 1.05 ? "text-red-700" : "text-neutral-900"}>
            {ukupno.toFixed(2)} KM
          </strong>{" "}
          <span className="text-neutral-500">/ cilj {cilj.toFixed(2)} KM</span>
        </p>

        <ul className="mt-4 flex flex-col gap-2">
          {redovi.map((r, i) => (
            <li key={r.productId} className="flex items-center gap-2">
              <span className="flex-1">{r.naziv}</span>
              <input
                type="number"
                step="0.1"
                min="0.1"
                value={r.kolicina}
                onChange={(e) => {
                  const kopija = [...redovi];
                  kopija[i] = { ...r, kolicina: Number(e.target.value) };
                  setRedovi(kopija);
                }}
                className="w-24 rounded border px-2 py-1"
                aria-label={`Količina za ${r.naziv}`}
              />
              <button
                type="button"
                onClick={() => setRedovi(redovi.filter((x) => x.productId !== r.productId))}
                className="text-sm text-red-700 underline"
              >
                Ukloni
              </button>
            </li>
          ))}
        </ul>

        <select
          className="mt-4 w-full rounded border px-2 py-2"
          value=""
          onChange={(e) => {
            const a = artikli?.find((x) => x.id === e.target.value);
            if (!a || redovi.some((r) => r.productId === a.id)) return;
            setRedovi([...redovi, { productId: a.id, naziv: a.name, kolicina: 1, cijena: a.price }]);
          }}
        >
          <option value="">+ Dodaj artikal…</option>
          {artikli?.map((a) => (
            <option key={a.id} value={a.id}>
              {a.name} — {a.price} KM/{a.unit}
            </option>
          ))}
        </select>

        <div className="mt-6 flex justify-end gap-3">
          <button type="button" onClick={onClose} className="px-4 py-2">
            Odustani
          </button>
          <button
            type="button"
            onClick={() => sacuvaj.mutate()}
            disabled={redovi.length === 0 || sacuvaj.isPending}
            className="rounded bg-green-700 px-4 py-2 text-white disabled:opacity-50"
          >
            {sacuvaj.isPending ? "Čuvam…" : "Sačuvaj"}
          </button>
        </div>
      </div>
    </div>
  );
}
```
